# resourceCache

Global cache management providing advanced cache control, LRU eviction, and batch operations for all [`resource`](/reference/resource/resource) instances.

## API

```typescript
import { resourceCache } from '@hellajs/resource';
import type { Resource, ResourceError } from '@hellajs/resource';

interface ResourceCache {
  readonly map: CacheMapView;
  readonly config: CacheConfig;
  setConfig(config: Partial<CacheConfig>): void;
  set<K, T>(key: K, data: T, cacheTime: number, staleTime?: number): K;
  get<T = unknown>(key: unknown): T | undefined;
  update<T>(key: unknown, updater: T | ((old: T | undefined) => T)): boolean;
  cleanup(): void;
  updateMultiple<T>(updates: Array<CacheUpdate<T>>): void;
  invalidate(key: unknown): void;
  invalidateMultiple(keys: unknown[]): void;
  invalidateByPrefix(prefix: string): number;
  invalidateByPattern(pattern: RegExp): number;
  invalidateAll(): number;
  invalidateResources(resources: Array<Pick<Resource<unknown>, 'invalidate'>>): void;
  isOnline(): boolean;
  onOnlineChange(callback: (online: boolean) => void): () => void;
  prefetch<T, K>(options: PrefetchOptions<T, K>): Promise<T>;
}

interface CacheConfig {
  maxSize?: number;
  enableLRU?: boolean;
}

interface CacheEntry<T> {
  data: T;
  timestamp: number;
  cacheTime: number;
  staleTime: number;
  lastAccess: number;
}

interface CacheUpdate<T> {
  key: unknown;
  updater: T | ((old: T | undefined) => T);
}

interface CacheMapView {
  readonly size: number;
  get(key: unknown): CacheEntry<unknown> | undefined;
  has(key: unknown): boolean;
  clear(): void;
}

interface PrefetchOptions<T, K> {
  fetcher: (key: K) => Promise<T>;
  key: K;
  cacheTime?: number;
  staleTime?: number;
  timeout?: number;
  abortSignal?: AbortSignal;
  retry?: number | boolean | ((failureCount: number, error: ResourceError) => boolean);
  retryDelay?: number | ((attempt: number, error: ResourceError) => number);
  deduplicate?: boolean;
}
```

### Read Methods

### `map`

Read-only flattened view over every fetcher-scoped cache. `get`, `has`, and `size` search across all scopes; `clear()` empties every scope.

```typescript
import { resourceCache } from '@hellajs/resource';

resourceCache.set('user:123', { name: 'John' }, 60000);

console.log('Cache entries:', resourceCache.map.size); // 1
const entry = resourceCache.map.get('user:123');
console.log('Cached data:', entry?.data); // { name: 'John' }
```

### `config`

Readonly access to current cache configuration settings.

```typescript
console.log(resourceCache.config); // { maxSize: 1000, enableLRU: true }
```

### `get`

Retrieves data from the cache by key. Plain objects and arrays compare **structurally** — an equal-shaped fresh object hits an entry stored with a different reference. Any other object shape (`Date`, `Map`, `Set`, class instances) compares by reference.

```typescript
resourceCache.set('user:123', { name: 'John' }, 60000);

// Get cached data
const user = resourceCache.get<{ name: string }>('user:123');
if (user) console.log('Cached user:', user.name);

// Object keys compare structurally — same shape hits
resourceCache.set({ team: 1 }, [{ id: 1, name: 'Ada' }], 60000);
const roster = resourceCache.get<{ id: number; name: string }[]>({ team: 1 });
```

### `isOnline`

Returns the current browser online status.

```typescript
console.log(resourceCache.isOnline() ? 'Online' : 'Offline - using cached data');
```

### Write Methods

### `setConfig`

Updates the global cache configuration with new settings.

```typescript
import { resourceCache } from '@hellajs/resource';

// Update cache configuration
resourceCache.setConfig({
  maxSize: 2000,      // Maximum cache entries
  enableLRU: true     // Enable LRU eviction
});

// Partial updates work too
resourceCache.setConfig({ maxSize: 5000 });
```

Invalid input throws: `config` must be an object, `maxSize` a non-negative number, `enableLRU` a boolean.

### `set`

Stores data in the cache with a time-to-live and optional stale time. The `cacheTime` parameter is required: omitting it throws. An explicit `0` stores nothing (documented no-op) but still returns the key.

```typescript
// Set with TTL (5 minutes) and stale time (30 seconds)
resourceCache.set('user:123', { name: 'John' }, 300000, 30000);

// Set with TTL only (staleTime defaults to 0 = always stale on next fetch)
resourceCache.set('config', { theme: 'dark' }, 60000);

// explicit cacheTime 0 → nothing is stored (no-op), key still returned
resourceCache.set('temp', { data: 'gone' }, 0);
```

Note: `staleTime` defaults to `0` here (always stale), which differs from [`resource()`](/reference/resource/resource) where `staleTime` defaults to `Infinity` (never stale).

### `update`

Updates existing cached data using an updater function or direct value.

```typescript
// Seed the cache entry first
resourceCache.set('user:123', { name: 'John' }, 60000);

const updated = resourceCache.update<{ name: string; lastSeen?: number }>('user:123', old => ({
  ...old,
  lastSeen: Date.now()
}));
console.log(updated); // true

// Update with direct value
resourceCache.update('user:123', { name: 'Johnny' });
```

### `updateMultiple`

Updates multiple cache entries in a batch operation.

```typescript
// Seed cache entries first
resourceCache.set('user:1', { name: 'Alice' }, 60000);
resourceCache.set('user:2', { name: 'Bob' }, 60000);

// Batch update multiple users
type UserRecord = { name: string; online?: boolean };

resourceCache.updateMultiple<UserRecord>([
  { key: 'user:1', updater: user => ({ ...user, online: true }) },
  { key: 'user:2', updater: user => ({ ...user, online: false }) }
]);
```

### `cleanup`

Performs cleanup of expired cache entries to free memory.

```typescript
// Periodic cleanup
setInterval(() => resourceCache.cleanup(), 60000);
```

### `prefetch`

Fetches via `fetcher(key)` and caches the result under the fetcher's own scope **without creating a resource**, returning the fetched data. Deduplicates, retries, and aborts like a [`resource()`](/reference/resource/resource) fetch; the `retry` predicate receives the failure count starting at `1` on the first failure. See [Prefetching](#prefetching) for the same-fetcher-reference requirement that lets a later `resource()` reuse the entry.

```typescript
import { resourceCache } from '@hellajs/resource';

const fetchUser = async (id: number) => (await fetch(`/api/users/${id}`)).json();

const user = await resourceCache.prefetch({ fetcher: fetchUser, key: 1, cacheTime: 60000 });
console.log(user); // cached under fetchUser's scope
```

### Invalidation Methods

### `invalidate`

Removes a single entry from the cache by key.

```typescript
import { resourceCache } from '@hellajs/resource';

resourceCache.set('user:123', { name: 'John' }, 60000);

// Remove specific cache entry
resourceCache.invalidate('user:123');

// Entry is now gone
console.log(resourceCache.get('user:123')); // undefined
```

### `invalidateMultiple`

Removes multiple entries from the cache by keys.

```typescript
// Seed cache entries first
resourceCache.set('user:1', { name: 'Alice' }, 60000);
resourceCache.set('posts:user:1', [{ id: 1 }], 60000);

resourceCache.invalidateMultiple(['user:1', 'posts:user:1']);
```

### `invalidateByPrefix`

Removes all cache entries whose keys start with the given prefix. Returns the number of entries invalidated. Matches string keys only — structurally-hashed object/array keys are opaque strings not designed for prefix matching.

```typescript
// Set up some cached data
resourceCache.set('user-profile', { name: 'John' }, 60000);
resourceCache.set('user-settings', { theme: 'dark' }, 60000);
resourceCache.set('session-token', { token: 'abc' }, 60000);

const count = resourceCache.invalidateByPrefix('user-');
console.log(count); // 2 (user-profile, user-settings)
```

### `invalidateByPattern`

Removes all cache entries whose keys match the given regular expression pattern. Returns the number of entries invalidated. Matches string keys only — structurally-hashed object/array keys are opaque strings not designed for pattern matching.

```typescript
// Set up some cached data
resourceCache.set('todos-1', { text: 'First' }, 60000);
resourceCache.set('todos-2', { text: 'Second' }, 60000);
resourceCache.set('posts-1', { title: 'Post' }, 60000);

const count = resourceCache.invalidateByPattern(/^todos-\d+$/);
console.log(count); // 2 (todos-1, todos-2)
```

### `invalidateAll`

Clears all cache entries. Returns the number of entries invalidated.

```typescript
const count = resourceCache.invalidateAll();
console.log(`Cleared ${count} entries`);
console.log(resourceCache.map.size); // 0
```

### `invalidateResources`

Immediately invalidates all provided resources by calling their invalidate methods.

```typescript
import { resourceCache, resource } from '@hellajs/resource';

const fetchUserPosts = (id: string) => fetch(`/api/users/${id}/posts`).then(r => r.json());

const userResource = resource(fetchUser);
const userPostsResource = resource(fetchUserPosts);

resourceCache.invalidateResources([userResource, userPostsResource]);
```

### Utility Methods

### `onOnlineChange`

Subscribes to browser online/offline status changes. Returns an unsubscribe function.

```typescript
const unsubscribe = resourceCache.onOnlineChange((online) => {
  if (online) console.log('Back online - refreshing data');
  else console.log('Gone offline - using cached data');
});
unsubscribe(); // Clean up subscription
```

## Basic Usage

```typescript
import { resourceCache } from '@hellajs/resource';

// Store an entry with a 5-minute TTL
resourceCache.set('user:123', { name: 'John' }, 300000);

// Read it back — a hit within the TTL
console.log(resourceCache.get('user:123')); // { name: 'John' }

// Remove it
resourceCache.invalidate('user:123');
```

## Key Concepts

### Automatic Cache Isolation

Each [`resource`](/reference/resource/resource) has its cache automatically scoped by its fetcher function. This means two resources with different fetchers will never collide, even if they produce the same cache key value. Resources that share the same fetcher function share a cache scope, which is correct for patterns like data transformation where multiple resources derive different views from the same cached data.

The `resourceCache` API provides a flattened view across all scopes: `get()`, `invalidate()`, and similar methods search across all resource scopes transparently. Manual cache entries created with `resourceCache.set()` live in a separate public scope.

### Key Equality

Within a scope, keys compare by value for primitives and **structurally** for plain objects and arrays: recursively, property order insensitive, array order significant, nested dates compared by their ISO representation. Two evaluations that build distinct-but-equal objects (`key: () => ({ ...filters() })`) hit the same cache and dedup entries. Every other object shape — a `Date`, `Map`, `Set`, or class instance used as or nested in a key — compares by reference. [`invalidateByPrefix`](#invalidatebyprefix) and [`invalidateByPattern`](#invalidatebypattern) match string keys only.

### TTL and Expiration

Cache entries have a time-to-live (TTL) that determines when they expire; expired entries are cleaned up lazily during cache operations, throttled to avoid performance impact.

### LRU Eviction Strategy

When the cache reaches its `maxSize` limit, the least-recently-used entries are evicted automatically — memory stays within configured bounds while frequently accessed data remains available.

### Prefetching

[`prefetch`](#prefetch) warms the cache ahead of need. Cache scope and dedup are keyed by **fetcher reference identity**, so a later [`resource()`](/reference/resource/resource) hits the prefetched entry only when it shares the *same fetcher function reference* and key. The URL overload `resource('url')` cannot share a prefetched entry — each call builds a fresh fetcher closure and thus a separate scope. Use a named fetcher function to make the scope shareable. `structuralSharing` is intentionally absent from `PrefetchOptions`: prefetch has no prior data to share against.

```typescript
import { resource, resourceCache } from '@hellajs/resource';

// A named fetcher so the cache scope is shareable.
const fetchUser = async (id: number) => (await fetch(`/api/users/${id}`)).json();

await resourceCache.prefetch({ fetcher: fetchUser, key: 1, cacheTime: 60000 });

// A later resource sharing the same fetchUser reference + key hits the prefetched entry:
const userResource = resource(fetchUser, { key: 1, cacheTime: 60000 });
```

## Important Considerations

### Efficient Cache Patterns

Use selective invalidation and batch operations for optimal performance.

```typescript
// Selective key invalidation instead of clearing entire cache
const clearUserData = (userId: number) => {
  resourceCache.invalidateMultiple([
    `user:${userId}`,
    `user:${userId}:posts`,
    `user:${userId}:settings`,
  ]);
};

// Batch updates to reduce overhead
const updateUserStatuses = (updates: Array<{id: number, online: boolean}>) => {
  resourceCache.updateMultiple(
    updates.map(({id, online}) => ({
      key: `user:${id}`,
      updater: (user: { online?: boolean } | undefined) => ({ ...user, online })
    }))
  );
};
```

### Consistent Key Design

Use hierarchical, predictable key patterns with typed template-literal helpers.

```typescript
type UserCacheKey = `user:${number}`;
type PostCacheKey = `user:${number}:posts:${number}`;

const getUserKey = (id: number): UserCacheKey => `user:${id}`;
const getPostKey = (userId: number, postId: number): PostCacheKey =>
  `user:${userId}:posts:${postId}`;

const userData = { name: 'John' };
resourceCache.set(getUserKey(123), userData, 60000);
const user = resourceCache.get(getUserKey(123));
```

### Production Memory Management

Monitor cache size via `resourceCache.map.size` against `resourceCache.config.maxSize`. The LRU eviction strategy (see [LRU Eviction Strategy](#lru-eviction-strategy) above) automatically removes old entries when the limit is exceeded — adjust `maxSize` with `setConfig()` based on available memory.
