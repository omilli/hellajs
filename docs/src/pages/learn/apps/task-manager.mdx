# Task Manager

A task manager using [`store`](/reference/store/store) for state, [`ForEach`](/reference/dom/foreach) for lists, and [`css`](/reference/css/css) for styling.

**Packages used:** [`@hellajs/store`](/reference/store), [`@hellajs/dom`](/reference/dom), [`@hellajs/css`](/reference/css), [`@hellajs/core`](/reference/core)

## Setup

```bash
npm install @hellajs/core @hellajs/dom @hellajs/store @hellajs/css
```

## Complete App

```tsx
import { signal, computed, effect } from '@hellajs/core';
import { mount, ForEach } from '@hellajs/dom';
import { store } from '@hellajs/store';
import { css } from '@hellajs/css';

type Filter = 'all' | 'active' | 'completed';

interface Task {
  id: number;
  text: string;
  done: boolean;
}

const app = store({
  tasks: [
    { id: 1, text: 'Learn HellaJS', done: false },
    { id: 2, text: 'Build a task manager', done: false },
    { id: 3, text: 'Ship to production', done: true }
  ] as Task[],
  filter: 'all' as Filter,
  nextId: 4
});

const newTask = signal('');
const editingId = signal<number | null>(null);
const editText = signal('');

const filtered = computed(() => {
  const tasks = app.tasks();
  const f = app.filter();
  if (f === 'active') return tasks.filter(t => !t.done);
  if (f === 'completed') return tasks.filter(t => t.done);
  return tasks;
});

const stats = computed(() => {
  const tasks = app.tasks();
  return {
    total: tasks.length,
    active: tasks.filter(t => !t.done).length,
    done: tasks.filter(t => t.done).length
  };
});

const addTask = () => {
  const text = newTask().trim();
  if (!text) return;
  app.update(d => {
    d.tasks.push({ id: d.nextId, text, done: false });
    d.nextId++;
  });
  newTask('');
};

const toggle = (id: number) => {
  app.update(d => {
    const t = d.tasks.find(t => t.id === id);
    if (t) t.done = !t.done;
  });
};

const removeTask = (id: number) => {
  app.update(d => {
    d.tasks = d.tasks.filter(t => t.id !== id);
  });
};

const startEdit = (id: number, text: string) => {
  editingId(id);
  editText(text);
};

const saveEdit = () => {
  const id = editingId();
  if (id == null) return;
  const text = editText().trim();
  if (!text) { removeTask(id); }
  else {
    app.update(d => {
      const t = d.tasks.find(t => t.id === id);
      if (t) t.text = text;
    });
  }
  editingId(null);
  editText('');
};

const clearDone = () => {
  app.update(d => { d.tasks = d.tasks.filter(t => !t.done); });
};

effect(() => {
  const s = stats();
  document.title = s.active > 0 ? `Tasks (${s.active})` : 'Tasks';
});

effect(() => {
  localStorage.setItem('tasks', JSON.stringify(app.tasks()));
});

// Styles
const styles = {
  container: css({ maxWidth: '36rem', margin: '2rem auto', padding: '0 1rem' }),
  header: css({ fontSize: '1.5rem', fontWeight: 'bold', marginBottom: '1rem' }),
  inputRow: css({ display: 'flex', gap: '0.5rem', marginBottom: '1rem' }),
  input: css({
    flex: 1, padding: '0.5rem', border: '1px solid #d1d5db', borderRadius: '0.375rem',
    fontSize: '0.875rem'
  }),
  btn: css({
    padding: '0.5rem 1rem', border: 'none', borderRadius: '0.375rem',
    cursor: 'pointer', fontSize: '0.875rem', fontWeight: 500
  }),
  filterRow: css({ display: 'flex', gap: '0.5rem', marginBottom: '1rem' }),
  task: css({
    display: 'flex', alignItems: 'center', gap: '0.5rem',
    padding: '0.5rem', borderBottom: '1px solid #e5e7eb'
  }),
  taskText: css({ flex: 1, fontSize: '0.875rem', cursor: 'pointer' }),
  done: css({ textDecoration: 'line-through', color: '#9ca3af' }),
  editInput: css({
    flex: 1, padding: '0.25rem 0.5rem', border: '1px solid #3b82f6',
    borderRadius: '0.25rem', fontSize: '0.875rem'
  }),
  stats: css({ fontSize: '0.75rem', color: '#6b7280', marginTop: '0.5rem' })
};

const TaskManager = () => (
  <div class={styles.container}>
    <h1 class={styles.header}>Tasks</h1>

    <div class={styles.inputRow}>
      <input
        bind:value={newTask}
        on:input={e => newTask((e.target as HTMLInputElement).value)}
        on:keydown={e => e.key === 'Enter' && addTask()}
        placeholder="What needs to be done?"
        class={styles.input}
      />
      <button on:click={addTask} class={styles.btn} style={{ background: '#3b82f6', color: '#fff' }}>
        Add
      </button>
    </div>

    <div class={styles.filterRow}>
      {(['all', 'active', 'completed'] as Filter[]).map(f => (
        <button
          bind:class={() => `${styles.btn} ${app.filter() === f ? 'bg-blue-500 text-white' : ''}`}
          style={{ background: app.filter() === f ? '#3b82f6' : '#e5e7eb', color: app.filter() === f ? '#fff' : '#374151' }}
          on:click={() => app.filter(f)}
        >
          {f.charAt(0).toUpperCase() + f.slice(1)}
          {() => {
            const s = stats();
            return f === 'all' ? ` (${s.total})` : f === 'active' ? ` (${s.active})` : ` (${s.done})`;
          }}
        </button>
      ))}
      {() => stats().done > 0 && (
        <button on:click={clearDone} class={styles.btn} style={{ background: '#fee2e2', color: '#dc2626' }}>
          Clear Done
        </button>
      )}
    </div>

    <ul>
      <ForEach each={filtered} use={(task) => (
        <li key={task.id} class={styles.task}>
          <input
            type="checkbox"
            checked={task.done}
            on:change={() => toggle(task.id)}
          />
          {() => editingId() === task.id ? (
            <input
              bind:value={editText}
              on:input={e => editText((e.target as HTMLInputElement).value)}
              on:keydown={e => { if (e.key === 'Enter') saveEdit(); if (e.key === 'Escape') { editingId(null); editText(''); } }}
              on:blur={saveEdit}
              class={styles.editInput}
            />
          ) : (
            <span
              bind:class={() => `${styles.taskText} ${task.done ? styles.done : ''}`}
              on:dblclick={() => startEdit(task.id, task.text)}
            >
              {task.text}
            </span>
          )}
          <button on:click={() => removeTask(task.id)} style={{ background: 'none', border: 'none', color: '#dc2626', cursor: 'pointer', fontSize: '0.75rem' }}>
            ✕
          </button>
        </li>
      )} />
    </ul>

    <div class={styles.stats}>
      {() => { const s = stats(); return `${s.active} remaining, ${s.done} completed`; }}
    </div>
  </div>
);

mount(TaskManager, '#app');
```

**Key patterns:**

- **[Store](/reference/store/store) for complex state**: Tasks, filter, and nextId managed in a single store with draft-based mutations
- **[Computed](/reference/core/computed) derivations**: `filtered` and `stats` derive from store state, no manual syncing
- **[ForEach](/reference/dom/foreach) with keys**: Efficient list rendering with key-based reconciliation for reorderings
- **Inline editing**: Separate [signals](/reference/core/signal) (`editingId`, `editText`) coordinate edit state across items
